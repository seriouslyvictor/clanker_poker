```python
"""
Integration tests for LiteLLM provider connectivity.

Each test is SKIPPED (not failed) when the corresponding env var is absent.
This allows CI to pass with zero API keys configured.

To run with real keys:
  cp .env.example .env
  # fill in your API keys
  uv run pytest tests/test_litellm_providers.py -v

Providers and model strings:
  OpenAI:    openai/gpt-4o                       (OPENAI_API_KEY)
  Anthropic: anthropic/claude-3-5-haiku-20241022  (ANTHROPIC_API_KEY)
  Google:    gemini/gemini-2.0-flash              (GEMINI_API_KEY)
  Ollama:    ollama_chat/llama3                   (no key — uses OLLAMA_API_BASE, default: http://localhost:11434)
"""
import os
import pytest
from litellm import completion


@pytest.mark.skipif(not os.getenv("OPENAI_API_KEY"), reason="OPENAI_API_KEY not set")
def test_openai_provider():
    response = completion(
        model="openai/gpt-4o",
        messages=[{"role": "user", "content": "ping"}],
        max_tokens=5,
    )
    assert response.choices[0].message.content is not None


@pytest.mark.skipif(not os.getenv("ANTHROPIC_API_KEY"), reason="ANTHROPIC_API_KEY not set")
def test_anthropic_provider():
    response = completion(
        model="anthropic/claude-3-5-haiku-20241022",
        messages=[{"role": "user", "content": "ping"}],
        max_tokens=5,
    )
    assert response.choices[0].message.content is not None


@pytest.mark.skipif(not os.getenv("GEMINI_API_KEY"), reason="GEMINI_API_KEY not set")
def test_gemini_provider():
    response = completion(
        model="gemini/gemini-2.0-flash",
        messages=[{"role": "user", "content": "ping"}],
        max_tokens=5,
    )
    assert response.choices[0].message.content is not None


def test_ollama_provider():
    """Ollama has no API key — skip if server not reachable on OLLAMA_API_BASE."""
    import httpx
    api_base = os.getenv("OLLAMA_API_BASE", "http://localhost:11434")
    try:
        httpx.get(f"{api_base}/api/tags", timeout=2.0)
    except Exception:
        pytest.skip("Ollama not reachable — install Ollama and run `ollama pull llama3` to enable")
    response = completion(
        model="ollama_chat/llama3",
        messages=[{"role": "user", "content": "ping"}],
        api_base=api_base,  # REQUIRED: Ollama is self-hosted, no default discovery
        max_tokens=5,
    )
    assert response.choices[0].message.content is not None
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| .env file → git history | A .env with real keys committed once is permanently in git history |
| os.getenv() → test assertions | If an API key is set but invalid, LiteLLM will return an error response (not exception in all cases) |
| litellm → external LLM APIs | Real money spent on API calls when tests run with real keys |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Information Disclosure | `backend/.env` in git | mitigate | `.gitignore` entry `backend/.env` created in Plan 01 Task 1. Verify in this plan's Task 1 before any other work. `git status` must never show `.env` as untracked. |
| T-03-02 | Information Disclosure | API keys in test output | mitigate | LiteLLM masks keys in its own logs. Tests only assert `response.choices[0].message.content is not None` — never print or log the key values. Test file uses `os.getenv("KEY")` only for existence check, not to log it. |
| T-03-03 | Denial of Service | LiteLLM test consuming API budget | accept | `max_tokens=5` cap on all test calls minimizes cost. Tests only run when dev explicitly provides keys. No CI runs these tests without keys (all skip). Accepted as normal dev expense. |
| T-03-04 | Spoofing | Ollama endpoint not authenticated | accept | Ollama is a local service on localhost:11434 by default. Phase 1 is local-only. Network-accessible Ollama is a Phase 4+ deployment concern. |
</threat_model>

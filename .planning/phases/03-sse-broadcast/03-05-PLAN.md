"""
Tests for backend/app/api/stream.py

Covers:
  STREAM-01: All clients receive identical game_state events (fan-out verified via broker)
  STREAM-02: Late joiners receive snapshot as first SSE event
  STREAM-03: X-Accel-Buffering: no header present; `: ping` heartbeat wired at 5s
  SC-4:      Monotonically increasing integer message IDs on every event
"""
import asyncio
import json
import pytest
import httpx
from httpx_sse import aconnect_sse
from fastapi.testclient import TestClient

from app.broadcast.broker import EventBroker
from app.main import app


# ---------------------------------------------------------------------------
# Test fixtures — inject stubs so no Redis is needed
# ---------------------------------------------------------------------------

SAMPLE_STATE_JSON = json.dumps({
    "phase": "flop",
    "pot": 100,
    "communityCards": [{"s": "♥", "r": "A"}, {"s": "♠", "r": "K"}, {"s": "♦", "r": "Q"}],
    "players": [],
    "showCards": False,
})


class StubRedis:
    """Minimal Redis stub: returns a preset snapshot for GET, no-ops for everything else."""

    def __init__(self, snapshot: bytes | None = None) -> None:
        self._snapshot = snapshot

    async def get(self, key: str) -> bytes | None:
        return self._snapshot

    async def set(self, key: str, value: str, ex: int | None = None) -> None:
        pass

    async def publish(self, channel: str, message: str) -> None:
        pass

    async def aclose(self) -> None:
        pass


@pytest.fixture(autouse=False)
def inject_stubs(monkeypatch):
    """
    Inject stub broker and Redis into app.state before each test.
    Restores originals after. Works without lifespan (no asyncio.create_task).
    """
    # Patch lifespan to a no-op so TestClient doesn't start the game loop or Redis connection
    from contextlib import asynccontextmanager

    @asynccontextmanager
    async def noop_lifespan(app):
        app.state.broker = EventBroker()
        app.state.redis_client = StubRedis(snapshot=SAMPLE_STATE_JSON.encode())
        yield

    monkeypatch.setattr(app, "router", app.router)
    original_lifespan = app.router.lifespan_context
    app.router.lifespan_context = noop_lifespan
    yield
    app.router.lifespan_context = original_lifespan


# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------

class TestSSEEndpoint:

    def test_no_buffering_header(self, inject_stubs):
        """STREAM-03: X-Accel-Buffering: no header is present on /api/stream responses."""
        client = TestClient(app)
        with client.stream("GET", "/api/stream") as response:
            assert response.headers.get("x-accel-buffering") == "no", (
                f"Expected x-accel-buffering: no, got: {response.headers.get('x-accel-buffering')}"
            )

    @pytest.mark.asyncio
    async def test_snapshot_sent_on_connect(self, inject_stubs):
        """
        STREAM-02: Late joiner receives snapshot as the first SSE event.
        The stub Redis returns SAMPLE_STATE_JSON as the snapshot.
        """
        async with httpx.AsyncClient(
            transport=httpx.ASGITransport(app=app), base_url="http://test"
        ) as client:
            async with aconnect_sse(client, "GET", "/api/stream") as event_source:
                events = []
                async for sse in event_source.aiter_sse():
                    events.append(sse)
                    if len(events) >= 1:
                        break

        assert len(events) >= 1, "Expected at least one SSE event (snapshot)"
        assert events[0].event == "game_state", (
            f"Expected event type 'game_state', got '{events[0].event}'"
        )
        data = json.loads(events[0].data)
        assert "phase" in data, f"Snapshot payload missing 'phase' field: {data}"
        assert data["phase"] == "flop", f"Expected phase 'flop', got '{data['phase']}'"

    @pytest.mark.asyncio
    async def test_message_id_monotonic(self, inject_stubs):
        """
        SC-4: Message IDs are monotonically increasing integers.
        Inject a second event via the broker after the snapshot, verify IDs increase.
        """
        # Arrange: put a live event into the broker after a short delay
        second_event_json = json.dumps({"phase": "turn", "pot": 200, "communityCards": [], "players": [], "showCards": False})

        async def push_event():
            await asyncio.sleep(0.05)
            broker = app.state.broker
            await broker.broadcast(second_event_json)

        asyncio.create_task(push_event())

        async with httpx.AsyncClient(
            transport=httpx.ASGITransport(app=app), base_url="http://test"
        ) as client:
            async with aconnect_sse(client, "GET", "/api/stream") as event_source:
                events = []
                async for sse in event_source.aiter_sse():
                    events.append(sse)
                    if len(events) >= 2:
                        break

        assert len(events) >= 2, "Expected snapshot + at least one live event"
        id_first = int(events[0].id)
        id_second = int(events[1].id)
        assert id_second > id_first, (
            f"Message IDs must be monotonically increasing: got {id_first}, {id_second}"
        )

    @pytest.mark.asyncio
    async def test_event_type_is_game_state(self, inject_stubs):
        """STREAM-01: SSE events use event type 'game_state' (not 'message' or empty)."""
        async with httpx.AsyncClient(
            transport=httpx.ASGITransport(app=app), base_url="http://test"
        ) as client:
            async with aconnect_sse(client, "GET", "/api/stream") as event_source:
                events = []
                async for sse in event_source.aiter_sse():
                    events.append(sse)
                    if len(events) >= 1:
                        break

        assert events[0].event == "game_state", (
            f"Expected event='game_state', got event='{events[0].event}'"
        )

    def test_health_endpoint_still_works(self, inject_stubs):
        """Regression: existing /health endpoint unaffected by Phase 3 changes."""
        client = TestClient(app)
        response = client.get("/health")
        assert response.status_code == 200
        assert response.json() == {"status": "ok"}
